# Creating a Text Form[](#creating-a-text-form "Link to this heading")

[`StringFormWithCallbacks`](../../reference/de/db/_autosummary/keysight.ads.de.db.StringFormWithCallbacks.md#keysight.ads.de.db.StringFormWithCallbacks "keysight.ads.de.db.StringFormWithCallbacks") is the class you use for creating a [text form](..%5C..%5C..%5C..%5C..%5C..%5C..%5Cads%5CContent%5Cads2026update2%5Cael%5Ccreate_text_form%28%29.md) for a component parameter.

Below is an example item definition of a component with an `InstSelectionForm` that provides a selection box in the [Edit Instance Parameters](..%5C..%5C..%5C..%5C..%5C..%5C..%5Cads%5CContent%5Cads2026update2%5Cusrguide%5CEditing_Component_Parameters.md) dialog.

```
# The following code assumes the item definition for the component is located in a cell called "text_form"
# and there is a symbol view for it.
# You can copy this code to a file called itemdef.py and place it in the text_form directory of your library.

from keysight.ads import de

_current_lpp_names = []

def layer_lpp_names(design: de.db_uu.Design) -> list[str]:
    """Return all the layer-purpose pairs in the library containing the design as "layer:purpose" strings."""
    from keysight.ads.de.experimental import _lpp as lpp

    lpp_db = lpp.LppDb(design.library)
    global _current_lpp_names
    _current_lpp_names = [
        f"{layer.layer_name}:{layer.purpose_name}" for layer in lpp_db.ordered_items if layer.library == design.library
    ]

    return _current_lpp_names

def validate_lpp_name(name: str) -> bool:
    """Check if the item is a valid layer-purpose pair."""
    global _current_lpp_names
    return name.strip('"') in _current_lpp_names

def layer_id_for_netlist(param: de.db.ParamItem, design: de.db_uu.Design) -> list[str]:
    """Return the Layer Id as netlist data for the Layer parameter."""
    assert isinstance(param, de.db.ParamItemString)
    assert ":" in param.value, "Expected value to contain a layer name and purpose."
    # Split the value into layer and purpose, stripping quotes
    # Example value: '"cond:drawing"'
    layer, purpose = [s.strip('"') for s in param.value.split(":")]
    # Get the layer ID and return it
    layer_id = de.db.LayerId.from_name(design, layer, purpose)
    return [str(layer_id.layer)]

def create_itemdef(cell: de.Cell) -> None:
    """Item definition for a text form that'll create a selection box in the edit instance parameters dialog."""
    form = de.db.StringFormWithCallbacks(
        "LayerTextForm",
        "LayerTextForm",
        "%d",
        "%v",
        "InstSelectionForm",
        layer_lpp_names,
        validate_lpp_name,
        layer_id_for_netlist,
    )

    # Constant form used later to illustrate how to force a parameter value to a form in the formset.
    na_form = de.db.ConstForm("Not Applicable", "NA", "N/A", "Not Applicable")

    library = cell.library
    library.forms.add(form)
    library.forms.add(na_form)
    # An InstSelectionForm is a dialog control that allows the user to select an item from a list of strings in the Edit Instance Parameters dialog.
    formset = de.db.Formset("InstSelectionForm", [form, na_form])
    library.formsets.add(formset)

    layer_param = de.db.ModelParam("Layer", "Layout layer where the artwork will be drawn.", formset)
    layer_param.default_value = de.db.ParamItemString("", form, '"cond:drawing"')
    layer_param.is_netlistable = True
    model_def = de.db.ModelDef(cell.cell_name, cell.cell_name)
    model_def.inst_name_prefix = "X"
    model_def.is_sub_design = False
    model_def.parameters = [layer_param]
    de.add_model_definition(library, model_def)

```

The following examples show how to call StringForm callbacks related to the options for that form.

```
def options_from_string_form_with_callbacks(design: de.db_uu.Design) -> None:
    """Retrieve the options from a StringFormWithCallbacks."""
    # Assuming schematic for example purposes
    assert design.is_schematic
    assert design.library.has_tech
    design.clear_design()

    # This example assumes create_itemdef has been called, there is a symbol view in the library,
    # and an instance is being placed into a design from the same library.
    text_item_inst = design.add_instance((f"{design.library.name}", "text_form", "symbol"), (0, 0))
    layer_param = text_item_inst.parameters["Layer"]
    assert isinstance(layer_param, de.db.ParamNonRepeated)
    layer_param_form = layer_param.form
    print(layer_param_form.name)
    assert layer_param_form.is_string_form(layer_param_form)
    assert isinstance(layer_param_form, de.db.StringFormWithCallbacks)

    # get_form_options_from_callback will call layer_lpp_names to get the options.
    options = layer_param_form.get_form_options_from_callback(design)
    assert len(options) > 0
    assert "cond:drawing" in options
    for option in options:
        print(f"layer:purpose: {option}")

def options_from_string_form_with_ael_callbacks(design: de.db_uu.Design) -> None:
    """Retrieve the options from a StringFormWithAELCallbacks."""
    # Assuming schematic for example purposes
    assert design.is_schematic
    assert design.library.has_tech
    design.clear_design()

    # Many components in ADS have parameters with string forms whose options are populated by AEL callbacks.
    # The MSUB component has several text form parameters, one of which is Cond1.
    msub = design.add_instance("ads_tlines:MSUB:symbol", (0, 0))
    cond1_param = msub.parameters["Cond1"]
    assert isinstance(cond1_param, de.db.ParamNonRepeated)
    cond1_param_form = cond1_param.form
    print(cond1_param_form.name)
    assert cond1_param_form.is_string_form(cond1_param_form)
    assert isinstance(cond1_param_form, de.db.StringFormWithAELCallbacks)

    options = cond1_param_form.get_form_options_from_callback(design)
    assert len(options) > 0
    assert "cond:drawing" in options
    for option in options:
        print(f"layer:purpose: {option}")

def data_strings_from_callback(design: de.db_uu.Design) -> None:
    """Retrieve the data strings from a StringFormWithCallbacks for a ParamItem."""
    from keysight.ads.de.experimental import _lpp as exp_lpp

    assert design.is_schematic
    assert design.library.has_tech
    design.clear_design()

    # This example assumes create_itemdef has been called, there is a symbol view in the library,
    # and an instance is being placed into a design from the same library.
    text_item_inst = design.add_instance((f"{design.library.name}", "text_form", "symbol"), (0, 0))
    layer_param = text_item_inst.parameters["Layer"]
    assert isinstance(layer_param, de.db.ParamNonRepeated)
    layer_param_form = layer_param.form
    assert isinstance(layer_param_form, de.db.StringFormWithCallbacks)
    lpp_db = exp_lpp.LppDb(design.library)
    items = lpp_db.items
    for i in items:
        print(f"lpp: {i.layer_name}:{i.purpose_name} - {i.layer_id}")

    lpp = lpp_db.items.find("cond:drawing")
    assert lpp is not None
    layer_param.value = '"cond:drawing"'
    # get_data_strings_from_callback_for_netlist will call layer_id_for_netlist to get the data strings.
    data_strings = layer_param_form.get_data_strings_from_callback_for_netlist(layer_param.item, design)

    assert len(data_strings) == 1
    assert de.db.LayerId(int(data_strings[0])) == lpp.layer_id

def validate(design: de.db_uu.Design) -> None:
    assert design.is_schematic
    assert design.library.has_tech
    design.clear_design()

    # This example assumes create_itemdef has been called, there is a symbol view in the library,
    # and an instance is being placed into a design from the same library.
    text_item_inst = design.add_instance((f"{design.library.name}", "text_form", "symbol"), (0, 0))
    layer_param = text_item_inst.parameters["Layer"]
    assert isinstance(layer_param, de.db.ParamNonRepeated)
    layer_param_form = layer_param.form
    assert isinstance(layer_param_form, de.db.StringFormWithCallbacks)

    layer_param_form.get_form_options_from_callback(design)

    # Validate the layer parameter with an invalid and a valid value.
    item = layer_param_form.validate_and_set_string_value(layer_param.item, '"not_found:drawing"')
    assert item is None and layer_param.value != '"not_found:drawing"'
    item = layer_param_form.validate_and_set_string_value(layer_param.item, '"silk_screen:drawing"')
    assert item is not None and layer_param.value == '"silk_screen:drawing"'
```
